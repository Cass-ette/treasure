> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 给 treasure 加中英双语 UI：模板/Python/JS 文案全走 gettext，颜色模式跟随语言，导航栏切换。

**Architecture:** Flask-Babel 管模板与 Python 文案（msgid 用中文原文）；渲染时把当前 locale 的 catalog 导出为 JSON 注入 `window.I18N` 供 JS 用 `t('中文键')` 取值；语言存 cookie（1 年）→ 回退 Accept-Language；`data-color-mode` 由 locale 决定初始值，localStorage 手动覆盖。

**Tech Stack:** Flask 2.3.3 / Python 3.9 / Flask-Babel 3.1.0 / Babel 2.14.0 / Jinja2 3.1

**Spec:** `docs/superpowers/specs/2026-08-18-i18n-design.md`

```python
"""Babel i18n 初始化：locale 探测 + JS 字典导出 + .mo 自动编译."""
from pathlib import Path

from flask import request
from flask_babel import Babel, get_translations

babel = Babel()

TRANSLATIONS_DIR = Path(__file__).parent / 'translations'
LOCALES = ('zh_CN', 'en_US')


def get_locale():
    """cookie 'locale' 优先；否则 Accept-Language（en* → en_US，其余 → zh_CN）。"""
    cookie = request.cookies.get('locale')
    if cookie == 'zh':
        return 'zh_CN'
    if cookie == 'en':
        return 'en_US'
    best = request.accept_languages.best_match(['zh', 'en'], default='zh')
    return 'en_US' if best == 'en' else 'zh_CN'


def _ensure_mo_compiled():
    """.po 比 .mo 新（或 .mo 缺失）时自动编译，免手工 pybabel compile。"""
    from babel.messages.pofile import read_po
    from babel.messages.mofile import write_mo

    for locale in LOCALES:
        po = TRANSLATIONS_DIR / locale / 'LC_MESSAGES' / 'messages.po'
        mo = TRANSLATIONS_DIR / locale / 'LC_MESSAGES' / 'messages.mo'
        if not po.exists():
            continue
        if mo.exists() and mo.stat().st_mtime >= po.stat().st_mtime:
            continue
        with open(po, 'rb') as f:
            catalog = read_po(f)
        with open(mo, 'wb') as f:
            write_mo(f, catalog)


def js_translations():
    """当前 locale 的 {msgid: msgstr} 字典，注入 window.I18N。"""
    _ensure_mo_compiled()
    cat = get_translations()
    if cat is None:
        return {}
    out = {}
    for msgid, msgstr in cat._catalog.items():
        if isinstance(msgid, str) and msgid:
            out[msgid] = msgstr or msgid
    return out


def init_babel(app):
    _ensure_mo_compiled()
    app.config['BABEL_TRANSLATION_DIRECTORIES'] = str(TRANSLATIONS_DIR)
    babel.init_app(app, locale_selector=get_locale)

    # 模板里可用 {{ js_translations() }}
    app.jinja_env.globals['js_translations'] = js_translations

    @app.context_processor
    def _inject_locale():
        return {'current_locale': get_locale()[:2]}
```

"""语言切换接口."""
from flask import Blueprint, request, jsonify

bp = Blueprint('locale', __name__)


@bp.route('/locale/set', methods=['POST'])
def set_locale():
    lang = request.form.get('lang')
    if lang not in ('zh', 'en'):
        return jsonify(ok=False, error='invalid lang'), 400
    resp = jsonify(ok=True)
    resp.set_cookie('locale', lang, max_age=365 * 24 * 3600, samesite='Lax')
    return resp
```

// i18n.js — 翻译 helper + 语言/颜色切换
window.t = function (k) { return (window.I18N || {})[k] || k; };

window.setLocale = function (lang) {
    fetch('/locale/set', {
        method: 'POST',
        headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
        body: 'lang=' + encodeURIComponent(lang)
    }).then(function (r) { return r.json(); })
      .then(function (d) { if (d.ok) location.reload(); });
};

// 颜色模式三态：follow / cn / intl
window.setColorMode = function (mode) {
    var el = document.documentElement;
    var lang = el.getAttribute('lang') === 'en-US' ? 'en' : 'zh';
    if (mode === 'follow') {
        localStorage.removeItem('colorMode');
        el.setAttribute('data-color-mode', lang === 'zh' ? 'cn' : 'intl');
    } else {
        localStorage.setItem('colorMode', mode);
        el.setAttribute('data-color-mode', mode);
    }
    updateColorLabel(mode);
};

function updateColorLabel(mode) {
    var label = document.getElementById('color-mode-label');
    if (!label) return;
    var names = { follow: t('跟随语言'), cn: t('红涨绿跌'), intl: t('绿涨红跌') };
    label.textContent = names[mode] || mode;
}

document.addEventListener('DOMContentLoaded', function () {
    var stored = localStorage.getItem('colorMode');
    updateColorLabel(stored || 'follow');
});
```

## 关键实现细节备忘

1. **`_` 的导入**：模板里 Jinja 全局自带（Flask-Babel 注入）；Python 路由文件里 `from flask_babel import gettext as _`
2. **模板里绝不能用 `{{ t(...) }}`**——`t` 是浏览器端函数，Jinja 渲染时 undefined 会 500。模板用 `{{ _('...') }}`，JS 用 `t('...')`
3. **f-string 改具名插值**：`f'基金 {code} 已存在'` → `_('基金 %(code)s 已存在', code=code)`
4. **pybabel extract 必须带 `-k t`**（配合 babel.cfg 的 `[javascript: ...]` 段才提取 JS 文案）
5. **.mo 不用手工编译**：`_ensure_mo_compiled()` 在 app 启动时自动做——但 flask_babel 按进程缓存已加载 catalog，**改了 .po 要重启 dev server 才生效**
6. **测试里设置语言**：`client.set_cookie('locale', 'en')`；测试需要 DB 表时**测试内显式 `db.create_all()`**（i18n_app fixture 不建表）
7. **JS 键与 .po 逐字一致**：迁移时用 grep 提取核对（Task 13 Step 3）
